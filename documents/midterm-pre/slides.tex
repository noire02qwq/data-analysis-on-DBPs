\section{Introduction \& Objectives}

\subsection{Background}
\begin{frame}{Background \& DBP Formation}
  \begin{block}{The Challenge}
    \begin{itemize}
        \item \textbf{Disinfection By-products (DBPs)}: Formed by reaction of disinfectants (e.g., Chlorine) with Natural Organic Matter (NOM).
        \item \textbf{Varieties}: Trihalomethanes (THMs), Haloacetic acids (HAAs), Haloacetonitriles (HANs).
        \item \textbf{Goal}: Use Deep Learning for real-time prediction in a simulated drinking water distribution network.
    \end{itemize}
  \end{block}

  \begin{block}{Formation Chemistry (Alkaline Conditions)}
    \begin{itemize}
        \item \textbf{Chloramine Formation}:
        $ \text{NH}_3 + \text{OCl}^- \rightarrow \text{NH}_2\text{Cl} + \text{OH}^- $
        \item \textbf{Mechanism}: Chloramines react with organic precursors to generate N-DBPs (e.g., Nitrosamines).
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Research Gap}
  \begin{itemize}
      \item \textbf{Dynamic Conditions}: Traditional methods (e.g., mechanistic models) struggle with real-time, non-linear dynamics.
      \item \textbf{Limit of Traditional ML}: Conventional machine learning (e.g., Decision Trees, SVM) often fails to capture complex temporal dependencies in time-series data.
      \item \textbf{Monitoring Limitations}: Lab analysis is infrequent and costly (low temporal resolution).
      \item \textbf{Opportunity}: High-frequency sensor data + Deep Learning for real-time, proactive control.
  \end{itemize}
\end{frame}

\subsection{Objectives}
\begin{frame}{Objectives}
  \begin{enumerate}
      \item \textbf{Development of Deep Learning Models}:
      \begin{itemize}
          \item Compare MLP, RNN, LSTM, and GRU architectures.
      \end{itemize}
      \item \textbf{Analyze Correlations (in progress)}:
      \begin{itemize}
          \item Investigate relationship between sensor data (TRC, pH, TOC) and DBP formation.
      \end{itemize}
      \item \textbf{Develop Prediction Framework (in progress)}:
      \begin{itemize}
          \item Establish a robust framework for real-time water quality monitoring.
      \end{itemize}
  \end{enumerate}
\end{frame}

\begin{frame}{Hyperparameter Optimization}
  \begin{block}{Bayesian Optimization}
      \begin{itemize}
          \item \textbf{Why?}: Deep learning models are sensitive to hyperparameters (e.g., learning rate, hidden size). Manual tuning is inefficient.
          \item \textbf{Process}: 
          \begin{enumerate}
              \item Build a probabilistic surrogate model (Gaussian Process) of the objective function.
              \item Use an acquisition function to select the next set of hyperparameters to evaluate.
              \item Balance exploration (trying new regions) and exploitation (refining good regions).
          \end{enumerate}
          \item \textbf{Result}: Found optimal configurations with fewer iterations than Grid Search.
      \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Algorithm Optimizing}
  \small
  \begin{block}{Concept: Normalized Rate of Change}
    Instead of predicting the absolute downstream concentration ($Y_{down}$), we predict the normalized rate of change:
    \[ \text{Rate} = \frac{Y_{\text{downstream}} - Y_{\text{upstream}}}{Y_{\text{upstream}}} \]
  \end{block}
  \begin{block}{Rationale}
      \begin{itemize}
          \item Absolute concentrations are dominated by the stable upstream value.
          \item The chemical transformation causes only small deviations.
          \item Predicting the "Rate" forces the model to focus on the \textit{kinetics} of the reaction occurring within the pipe, rather than just memorizing the input level.
      \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Algorithm Optimizing}
  \footnotesize
  \begin{block}{Problem: TRC Behavior}
      \begin{itemize}
          \item Total Residual Chlorine (TRC) is the primary disinfectant and decays rapidly.
          \item Its behavior is distinct from stable parameters like Conductivity or TOC.
          \item In a combined multi-output model, TRC prediction accuracy often suffers.
      \end{itemize}
  \end{block}
  \begin{block}{Solution: Decoupled Modeling}
      \begin{itemize}
          \item \textbf{TRC Model}: Dedicated exclusively to predicting TRC.
          \item \textbf{Other Parameter Model}: Predicts all other parameters (pH, TOC, etc.).
          \item Allows for specialized architecture and hyperparameter tuning for the most critical parameter.
      \end{itemize}
  \end{block}
\end{frame}

\subsection{Conclusion}
\begin{frame}{Conclusion \& Future Work}
  \footnotesize
  \begin{block}{Key Findings}
    \begin{itemize}
        \item \textbf{General Performance}: Neural Networks fit the water quality data patterns effectively.
        \item \textbf{Comparison}: GRU and LSTM are superior, validating the need for temporal modeling in WDS.
        \item \textbf{Regression Strategy}:
        \begin{itemize}
            \item \textbf{TRC}: Best predicted using "Rate" regression (LSTM-Rate). Chemical decay is the dominant factor.
            \item \textbf{pH/TOC}: Best predicted using "Value" regression (GRU-Value).
        \end{itemize}
    \end{itemize}
  \end{block}

  \begin{block}{Future Work}
    \begin{itemize}
        \item \textbf{Second-Stage Model}: Map predicted water quality references (TRC, pH, TOC) to actual DBP concentrations (THMs, HAAs).
    \end{itemize}
  \end{block}
