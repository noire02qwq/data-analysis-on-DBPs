\section{Introduction \& Objectives}

\subsection{Background \& Chemistry}
\begin{frame}{Background \& DBP Formation}
  \begin{block}{The Challenge}
    \begin{itemize}
        \item \textbf{Disinfection By-products (DBPs)}: Formed by reaction of disinfectants (e.g., Chlorine) with Natural Organic Matter (NOM).
        \item \textbf{Health Risk}: Carcinogenic potential (e.g., Nitrosamines).
        \item \textbf{Goal}: Use Deep Learning for real-time prediction in Water Distribution Systems (WDS).
    \end{itemize}
  \end{block}

  \begin{block}{Chemistry (Alkaline Conditions)}
    \begin{itemize}
        \item \textbf{Chloramine Formation}:
        $ \text{NH}_3 + \text{OCl}^- \rightarrow \text{NH}_2\text{Cl} + \text{OH}^- $
        \item \textbf{Mechanism}: Chloramines + Organic Precursors $\xrightarrow{O_2}$ N-DBPs.
    \end{itemize}
  \end{block}
\end{frame}

\subsection{Objectives}
\begin{frame}{Objectives}
  \begin{enumerate}
      \item \textbf{Evaluate Deep Learning Models}:
      \begin{itemize}
          \item Compare MLP, RNN, LSTM, and GRU architectures.
      \end{itemize}
      \item \textbf{Analyze Correlations}:
      \begin{itemize}
          \item Investigate relationship between sensor data (TRC, pH, TOC) and DBP formation.
      \end{itemize}
      \item \textbf{Develop Prediction Framework}:
      \begin{itemize}
          \item Establish a robust framework for real-time water quality monitoring.
      \end{itemize}
  \end{enumerate}
\end{frame}

\begin{frame}{Data Acquisition}
  \begin{block}{Sensor Network}
      \begin{itemize}
          \item \textbf{Comprehensive Monitoring}: Sensors are deployed at all key stages: Disinfection Tank (DT), Retention Tank (RT), and Pipelines (PPL1/PPL2).
          \item \textbf{Key Parameters}:
          \begin{itemize}
              \item \textbf{Core}: Total Residual Chlorine (TRC), pH, Conductivity.
              \item \textbf{Advanced}: fDOM, Dissolved Oxygen (DO), Total Organic Carbon (TOC), Dissolved Organic Carbon (DOC).
          \end{itemize}
          \item \textbf{Frequency}: High-resolution data collected at 5-minute intervals.
      \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Data Preprocessing: Imputation}
  \begin{block}{Challenge: Missing Values}
      \begin{itemize}
          \item Due to the shared sensor setup, PPL1 and PPL2 have alternating blocks of missing data (30-minute gaps).
          \item Standard mean imputation would reduce variance and distort the signal.
      \end{itemize}
  \end{block}
  \begin{block}{Solution: Stochastic Interpolation}
      \begin{itemize}
          \item \textbf{Step 1}: Linear interpolation between known points.
          \item \textbf{Step 2}: Injection of random noise matching the local variance of the signal.
          \item \textbf{Result}: A continuous time-series that preserves the statistical properties of the water quality dynamics.
      \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Data Preprocessing: Imputation (Visual)}
  \centering
  \includegraphics[width=0.9\textwidth]{../attachment/imputation.png}
\end{frame}

\begin{frame}{Data Preprocessing: Time Alignment}
  \begin{block}{Challenge: Hydraulic Delays}
      \begin{itemize}
          \item Water takes time to travel from the tanks to the pipelines.
          \item Direct correlation of timestamps would lead to incorrect cause-and-effect mapping.
      \end{itemize}
  \end{block}
  \begin{block}{Solution: Lag Correction}
      \begin{itemize}
          \item Calculated the hydraulic residence time for each segment.
          \item Shifted the input data series (DT/RT) forward in time to align with the corresponding output data (PPL1/PPL2).
          \item Ensures the model learns the transformation of the \textit{same} water parcel.
      \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Training Process}
  \begin{block}{Optimization Setup}
      \begin{itemize}
          \item \textbf{Loss Function}: Mean Squared Error (MSE) - penalizes larger errors, ideal for regression.
          \item \textbf{Optimizer}: Adam - combines momentum and adaptive learning rates for faster convergence.
      \end{itemize}
  \end{block}
  \begin{block}{Regularization}
      \begin{itemize}
          \item \textbf{Dropout}: Randomly deactivates neurons during training to prevent co-adaptation.
          \item \textbf{Early Stopping}: Halts training when validation loss stops improving to prevent overfitting.
      \end{itemize}
  \end{block}

\begin{frame}{Hyperparameter Optimization}
  \begin{block}{Bayesian Optimization}
      \begin{itemize}
          \item \textbf{Why?}: Deep learning models are sensitive to hyperparameters (e.g., learning rate, hidden size). Manual tuning is inefficient.
          \item \textbf{Process}: 
          \begin{enumerate}
              \item Build a probabilistic surrogate model (Gaussian Process) of the objective function.
              \item Use an acquisition function to select the next set of hyperparameters to evaluate.
              \item Balance exploration (trying new regions) and exploitation (refining good regions).
          \end{enumerate}
          \item \textbf{Result}: Found optimal configurations with fewer iterations than Grid Search.
      \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Strategy 1: Normalized Rate of Change}
  \begin{block}{Concept}
    Instead of predicting the absolute downstream concentration ($Y_{down}$), we predict the normalized rate of change:
    \[ \text{Rate} = \frac{Y_{\text{downstream}} - Y_{\text{upstream}}}{Y_{\text{upstream}}} \]
  \end{block}
  \begin{block}{Rationale}
      \begin{itemize}
          \item Absolute concentrations are dominated by the stable upstream value.
          \item The chemical transformation causes only small deviations.
          \item Predicting the "Rate" forces the model to focus on the \textit{kinetics} of the reaction occurring within the pipe, rather than just memorizing the input level.
      \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Strategy 2: Decoupled TRC Model}
  \begin{block}{The Problem with TRC}
      \begin{itemize}
          \item Total Residual Chlorine (TRC) is the primary disinfectant and decays rapidly.
          \item Its behavior is distinct from stable parameters like Conductivity or TOC.
          \item In a combined multi-output model, TRC prediction accuracy often suffers.
      \end{itemize}
  \end{block}
  \begin{block}{The Solution}
      \begin{itemize}
          \item \textbf{Model A}: Dedicated exclusively to predicting TRC.
          \item \textbf{Model B}: Predicts all other parameters (pH, TOC, etc.).
          \item Allows for specialized architecture and hyperparameter tuning for the most critical parameter.
      \end{itemize}
  \end{block}
\end{frame}
