\section{Introduction \& Objectives}

\subsection{Background}
\begin{frame}{Background \& DBP Formation}
  \begin{block}{The Challenge}
    \begin{itemize}
        \item \textbf{Disinfection By-products (DBPs)}: Formed by the reaction of disinfectants (e.g., Chlorine) with \textbf{Natural Organic Matter (NOM)}.
        \item \textbf{Health Concerns}: Potential carcinogenic properties (e.g., \textbf{THMs, HAAs, Nitrosamines}).
        \item \textbf{Goal}: Leverage high-frequency sensor data (\textbf{TRC, pH, TOC}) and Machine Learning for real-time predictive control.
    \end{itemize}
  \end{block}

  \begin{block}{Chemistry in Alkaline Conditions}
    \begin{itemize}
        \item $ \text{NH}_3 + \text{OCl}^- \rightarrow \textbf{NH}_2\textbf{Cl} + \text{OH}^- $
        \item Chloramines react with organic nitrogen precursors (secondary amines) leading to \textbf{N-DBPs} formation.
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Research Gap \& Limitations of Traditional Approaches}
  \begin{block}{Methodological Gaps}
    \begin{itemize}
        \item \textbf{Complex Kinetics}: Traditional \textbf{mechanistic physical models} struggle with highly dynamic, non-linear \textbf{hydraulic conditions} in real distribution networks.
        \item \textbf{Infrequent Sampling}: \textbf{Lab testing} provides extremely low temporal resolution and incurs high operational costs.
    \end{itemize}
  \end{block}
  \begin{block}{The Deep Learning Opportunity}
    \begin{itemize}
        \item Capable of autonomously capturing intricate \textbf{temporal dependencies} from \textbf{multi-variate continuous time-series data} without explicit manual feature shifting or strict physical assumptions.
    \end{itemize}
  \end{block}
\end{frame}

\subsection{Concepts}
\begin{frame}{Fine-Tuning Strategy for Reactor Generalization}
  \small
  \begin{block}{The Need for Transferability in Environmental Engineering}
      \begin{itemize}
          \item Real-world \textbf{water distribution systems} span vastly different environmental conditions (e.g., \textbf{temperature shifts, distinct water sources}).
          \item Training entirely new models for every condition is computationally inefficient and ignores the shared physical similarities between \textbf{reactors}.
      \end{itemize}
  \end{block}
  \begin{block}{Exploring Reactor Generalization}
      \begin{itemize}
          \item We introduce \textbf{Fine-Tuning Techniques} to systematically explore how effectively learned \textbf{chemical kinetics} can be generalized.
          \item This allows us to transfer core predictive knowledge from a \textbf{baseline reactor environment} to new, similar \textbf{physical reactors} operating under varied boundary conditions.
      \end{itemize}
  \end{block}
\end{frame}

\subsection{Objectives}
\begin{frame}{Objectives}
  \begin{block}{Core Research Goals}
    \begin{enumerate}
        \item \textbf{Evaluate Predictive Models}: Compare standard GBDTs (\textbf{XGBoost, LightGBM, CatBoost}) against advanced Neural Networks (\textbf{MLP, RNN, LSTM, GRU, Transformer, Mamba}) on complex continuous \textbf{water quality time-series data}.
        \item \textbf{Investigate Fine-Tuning Generalizability}: Test the transferability of learned kinetics across varying temperatures ($29^{\circ}\text{C}$ to $35^{\circ}\text{C}$) and distinct source \textbf{water distributions} (CAWW vs. LSWW).
        \item \textbf{Develop a Production GUI}: Package the machine learning backend into a user-friendly frontend application for \textbf{operational facility management}.
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}{Hydraulic Retention Time (HRT) \& Temporal Dynamics}
  \footnotesize
  \begin{block}{Physical Reactor Dynamics}
      \begin{itemize}
          \item In real physical pipelines, water physically takes hours to transit from the upstream \textbf{Disinfection Tank (DT)} to the downstream measurement points (\textbf{PPL1/PPL2}).
          \item This transit duration is known as the \textbf{Hydraulic Retention Time (HRT)}.
          \item \textbf{Continuous Reactions}: Chemical reactions (like \textbf{chlorine decay} and \textbf{DBP formation}) occur continuously along this spatial and temporal transit.
      \end{itemize}
  \end{block}
  \begin{block}{Modeling the Water Parcel}
      \begin{itemize}
          \item Traditional static tabular modeling ignores flow dynamics. Our \textbf{time-series sliding window approach} inherently mirrors the \textbf{HRT}. It allows the model to dynamically track the aging and reacting \textbf{"water parcel"} as it moves through the pipe network.
      \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Model Architectures Overview}
  \begin{block}{GBDTs (Single-Output Regressors)}
      \begin{itemize}
          \item \textbf{XGBoost, LightGBM, CatBoost}: Fast, tree-based ensemble learners utilizing \textbf{decision boundaries} and \textbf{histogram optimization}.
      \end{itemize}
  \end{block}
  \begin{block}{Neural Networks (Multi-Output Regressors)}
      \begin{itemize}

          \item \textbf{Transformer}: Multi-head global \textbf{self-attention} architectures.
          \item \textbf{Mamba}: \textbf{State Space Model (SSM)} enabling linear-time sequence processing.
      \end{itemize}
  \end{block}

\begin{frame}{Why \textbf{Transformer} is Suitable for Water Reactors}
  \small
  \begin{block}{The Attention Mechanism}
      \begin{equation*}
          \scalebox{0.8}{$\text{Attention}(Q,K,V) = \text{softmax}\left(\frac{QK^T}{\sqrt{d_k}}\right)V$}
      \end{equation*}
  \end{block}
  \begin{block}{Environmental Engineering Context}
      \begin{itemize}
          \item \textbf{Cyclical Operations}: \textbf{Water distribution systems} operate on highly periodic and \textbf{cyclical hydraulic schedules}.
          \item \textbf{Intrinsic Matching}: Historical data generated at identical pipeline locations during similar cycles exhibit immense \textbf{structural similarity}. 
          \item \textbf{Global Perception}: The \textbf{Attention mechanism} calculates correlation weights ($QK^T$) across the entire historical sequence simultaneously, allowing the model to naturally "attend" to these \textbf{recurring cyclical physical states} far better than strictly sequential models.
      \end{itemize}
  \end{block}
\end{frame}

\section{Results \& Conclusion}

\subsection{Theoretical Analysis}
\begin{frame}{Chemical Kinetics: First-Order Decay in Pipelines}
  \footnotesize
  \begin{block}{The Chemistry of Chlorine Transport}
      \begin{itemize}
          \item The decay of \textbf{Total Residual Chlorine (TRC)} from upstream (\textbf{DT/RT}) to downstream (\textbf{PPL1/PPL2}) follows a \textbf{first-order reaction}:
          \begin{equation*}
              \scalebox{0.8}{$C_t = C_0 e^{-kt}$}
          \end{equation*}
      \end{itemize}
  \end{block}
  \begin{block}{Dynamic Reactor Conditions}
      \begin{itemize}
          \item The \textbf{rate constant ($k$)} is dynamically influenced by real-time \textbf{environmental conditions} (\textbf{temperature, pH, flow}).
          \item This creates a continuous, non-linear exponential decay along the physical pipe network.
      \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Algorithm Expressiveness: GBDT vs Neural Networks}
  \small
  \begin{block}{\textbf{Neural Networks}: Continuous Mathematics}
      \begin{itemize}
          \item \textbf{Neural Networks} process inputs through multiple \textbf{nested non-linear activation functions} (e.g., Sigmoid, ReLU). 
          \item This continuous mathematical architecture natively maps to, and inherently approximates, smooth \textbf{continuous functional transformations} like true \textbf{exponential chemical decay}.
      \end{itemize}
  \end{block}
  \begin{block}{\textbf{GBDTs}: Discontinuous Step Functions}
      \begin{itemize}
          \item \textbf{Tree-based GBDTs} operate via greedy variance reduction, constructing orthogonal, discontinuous step functions.
          \item They partition the real-number data space into rigid "boxes." Consequently, they lack the \textbf{continuous functional expression power} necessary to physically extrapolate smooth \textbf{environmental reaction kinetics} beyond historically seen discrete thresholds.
      \end{itemize}
  \end{block}
\end{frame}

  \begin{block}{Environmental Variables Perception}
      \footnotesize
      \textbf{Partial Fine-Tuning} demonstrates the best robust generalization across \textbf{reactor conditions}. Each neural layer acts as a perceptron for specific \textbf{environmental conditions} (e.g., \textbf{pH, conductivity}, as well as some unmeasured physical parameters or substance concentrations). By freezing core layers and retraining only specific shallow neurons, we isolate the "\textbf{temperature}" shift without corrupting the fundamental \textbf{physical and chemical relationships} already learned by the network.
  \end{block}

\subsection{Conclusion}
\begin{frame}{Conclusion}
  \begin{block}{Key Engineering Takeaways}
      \begin{itemize}
          \item \textbf{Kinetic Modeling}: \textbf{Neural Networks}, leveraging nested \textbf{activation functions}, are fundamentally better equipped than \textbf{GBDTs} to capture continuous \textbf{environmental transformations} like \textbf{first-order chemical decay} in water networks.
          \item \textbf{Reactor Generalization}: \textbf{Transformer} architectures demonstrated exceptional transferability via \textbf{Partial Fine-Tuning}, successfully isolating \textbf{environmental factors} like \textbf{temperature shifts} across distinct \textbf{reactors}.
          \item \textbf{Industrial Readiness}: The developed \textbf{GUI application} successfully bridges advanced \textbf{kinetic AI modeling} with accessible industrial operational practice.
      \end{itemize}
  \end{block}
\end{frame}
